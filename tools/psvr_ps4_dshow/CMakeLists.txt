cmake_minimum_required(VERSION 3.20)
project(psvr_ps4_dshow LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "PoC requires Windows SDK and MSVC; installs nothing")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "PoC ABI is x64 only")
endif()
add_library(psvr_ps4_dshow SHARED backend.cpp)
target_compile_features(psvr_ps4_dshow PRIVATE cxx_std_17)
target_compile_definitions(psvr_ps4_dshow PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(psvr_ps4_dshow PRIVATE /W4 /EHsc /permissive-)
target_link_libraries(psvr_ps4_dshow PRIVATE ole32 oleaut32 strmiids)
set_property(TARGET psvr_ps4_dshow PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(PSVR_DSHOW_OFFLINE_TESTS "Compile fake COM fixtures (no device enumeration)" ON)
if(PSVR_DSHOW_OFFLINE_TESTS)
  enable_testing()
  add_executable(psvr_ps4_dshow_offline offline_tests.cpp)
  target_compile_features(psvr_ps4_dshow_offline PRIVATE cxx_std_17)
  target_compile_definitions(psvr_ps4_dshow_offline PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(psvr_ps4_dshow_offline PRIVATE /W4 /EHsc /permissive-)
  target_link_libraries(psvr_ps4_dshow_offline PRIVATE ole32 oleaut32 strmiids)
  set_property(TARGET psvr_ps4_dshow_offline PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME dshow_fake_com COMMAND psvr_ps4_dshow_offline)
  set_tests_properties(dshow_fake_com PROPERTIES TIMEOUT 15)
endif()
